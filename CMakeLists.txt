cmake_minimum_required(VERSION 3.24)
project(Citadel LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
include(FetchContent)
option(BUILD_TESTING "Build gameplay regression tests" OFF)
set(GLFW_BUILD_DOCS OFF CACHE BOOL "" FORCE)
set(GLFW_BUILD_EXAMPLES OFF CACHE BOOL "" FORCE)
set(GLFW_BUILD_TESTS OFF CACHE BOOL "" FORCE)
set(GLFW_INSTALL OFF CACHE BOOL "" FORCE)
FetchContent_Declare(glfw GIT_REPOSITORY https://github.com/glfw/glfw.git GIT_TAG 3.3.10 GIT_SHALLOW TRUE)
FetchContent_Declare(glm GIT_REPOSITORY https://github.com/g-truc/glm.git GIT_TAG 1.0.3 GIT_SHALLOW TRUE)
FetchContent_Declare(glad GIT_REPOSITORY https://github.com/dav1dde/glad.git GIT_TAG v2.0.8 GIT_SHALLOW TRUE SOURCE_SUBDIR cmake)
FetchContent_Declare(imgui GIT_REPOSITORY https://github.com/ocornut/imgui.git GIT_TAG v1.92.9b GIT_SHALLOW TRUE)
FetchContent_Declare(stb URL https://github.com/nothings/stb/archive/2c980bb59875b0d32144a71867fbdebb2f77cd20.tar.gz DOWNLOAD_EXTRACT_TIMESTAMP TRUE)
FetchContent_MakeAvailable(glfw glm glad imgui stb)
glad_add_library(citadel_gl STATIC REPRODUCIBLE API gl:core=3.3)
add_library(citadel_game STATIC src/game/Game.cpp src/game/World.cpp src/game/Exploration.cpp src/Localization.cpp)
target_include_directories(citadel_game PUBLIC src)
add_executable(Citadel src/main.cpp src/Engine.cpp src/Renderer.cpp src/HeldEquipment.cpp src/Sound.cpp
  ${imgui_SOURCE_DIR}/imgui.cpp ${imgui_SOURCE_DIR}/imgui_draw.cpp
  ${imgui_SOURCE_DIR}/imgui_tables.cpp ${imgui_SOURCE_DIR}/imgui_widgets.cpp
  ${imgui_SOURCE_DIR}/backends/imgui_impl_glfw.cpp ${imgui_SOURCE_DIR}/backends/imgui_impl_opengl3.cpp)
target_include_directories(Citadel PRIVATE src ${imgui_SOURCE_DIR} ${imgui_SOURCE_DIR}/backends ${stb_SOURCE_DIR})
target_link_libraries(Citadel PRIVATE citadel_game glfw glm::glm citadel_gl)
target_compile_definitions(Citadel PRIVATE CITADEL_ASSET_DIR="${CMAKE_CURRENT_SOURCE_DIR}/assets")
if(WIN32)
  enable_language(RC)
  configure_file(src/Citadel.rc.in ${CMAKE_CURRENT_BINARY_DIR}/Citadel.rc @ONLY)
  target_sources(Citadel PRIVATE ${CMAKE_CURRENT_BINARY_DIR}/Citadel.rc)
  target_link_libraries(Citadel PRIVATE winmm imm32)
endif()
if(MSVC)
  target_compile_options(citadel_game PRIVATE /utf-8 /EHsc)
  target_compile_options(Citadel PRIVATE /utf-8 /EHsc)
endif()
set_target_properties(Citadel PROPERTIES RUNTIME_OUTPUT_DIRECTORY ${CMAKE_BINARY_DIR})
add_custom_command(TARGET Citadel POST_BUILD COMMAND ${CMAKE_COMMAND} -E copy_directory ${CMAKE_CURRENT_SOURCE_DIR}/assets $<TARGET_FILE_DIR:Citadel>/assets)
if(BUILD_TESTING)
  include(CTest)
  FetchContent_Declare(Catch2 GIT_REPOSITORY https://github.com/catchorg/Catch2.git GIT_TAG v3.15.3 GIT_SHALLOW TRUE)
  FetchContent_MakeAvailable(Catch2)
  add_executable(citadel_tests tests/test_game.cpp)
  target_link_libraries(citadel_tests PRIVATE citadel_game Catch2::Catch2WithMain)
  target_compile_definitions(citadel_tests PRIVATE CITADEL_TEST_ASSET_DIR="${CMAKE_CURRENT_SOURCE_DIR}/assets")
  if(MSVC)
    target_compile_options(citadel_tests PRIVATE /utf-8 /EHsc)
  endif()
  add_test(NAME citadel_gameplay COMMAND citadel_tests)
endif()
